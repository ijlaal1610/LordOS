# config.txt — LordOS for Raspberry Pi 5 (and Pi 4, where possible). See docs/RASPBERRY.md.
# Read by the board firmware before the kernel. It can be edited with the card in another machine.

[all]
arm_64bit=1
# initramfs with the same name as the kernel (kernel_2712.img → initramfs_2712, kernel8.img → initramfs8)
auto_initramfs=1
# Full KMS: vc4 drives HDMI (the display) and v3d the 3D GPU. lord-server uses KMS without libdrm.
dtoverlay=vc4-kms-v3d
max_framebuffers=2
# the firmware does not touch the modes: the kernel picks them (KMS)
disable_fw_kms_setup=1
disable_overscan=1
# DSI displays and official cameras, if present
display_auto_detect=1
camera_auto_detect=1
dtparam=audio=on

[pi5]
# NVMe through the M.2 HAT (or the official base): external PCIe on. Gen 3 doubles the speed;
# officially it is gen 2 and some drives fail at gen 3: if the system does not boot from the NVMe or
# there are PCIe errors in `dmesg`, comment out the gen 3 line.
dtparam=pciex1
dtparam=pciex1_gen=3
# Active Cooler fan: driven by the kernel (cooling_fan in the device tree), nothing to set

[pi4]
# Pi 4: best effort (see docs/RASPBERRY.md)
arm_boost=1

[all]
