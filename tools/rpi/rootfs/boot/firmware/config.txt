# config.txt — LordOS para Raspberry Pi 5 (y Pi 4, en lo posible). Ver docs/RASPBERRY.md.
# Lo lee el firmware de la placa antes del kernel. Se puede editar con la tarjeta en otra máquina.

[all]
arm_64bit=1
# initramfs con el mismo nombre que el kernel (kernel_2712.img → initramfs_2712, kernel8.img → initramfs8)
auto_initramfs=1
# KMS completo: vc4 maneja HDMI (la pantalla) y v3d la GPU 3D. lord-server usa KMS sin libdrm.
dtoverlay=vc4-kms-v3d
max_framebuffers=2
# el firmware no toca los modos: los elige el kernel (KMS)
disable_fw_kms_setup=1
disable_overscan=1
# pantallas DSI y cámaras oficiales, si hay
display_auto_detect=1
camera_auto_detect=1
dtparam=audio=on

[pi5]
# NVMe por el M.2 HAT (o la base oficial): PCIe externo encendido. Gen 3 duplica la velocidad;
# oficialmente es gen 2 y algunos discos fallan en gen 3: si el sistema no arranca desde el NVMe o
# hay errores de PCIe en `dmesg`, comentar la línea de gen 3.
dtparam=pciex1
dtparam=pciex1_gen=3
# ventilador del Active Cooler: lo maneja el kernel (cooling_fan en el device tree), nada que poner

[pi4]
# Pi 4: mejor esfuerzo (ver docs/RASPBERRY.md)
arm_boost=1

[all]
